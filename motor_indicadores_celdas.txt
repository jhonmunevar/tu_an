# ===================== 5. FUNCIONES DEL MOTOR =====================
import functools

def normalizar(col, tipo):
    """CODIGO numérico: '01' y '1' quedan iguales (hallazgo calidad). NUMERICO/MONETARIO: decimal."""
    if tipo in ("NUMERICO", "MONETARIO"):
        return F.trim(col.cast("string")).cast("decimal(38,4)")
    c = F.trim(col.cast("string"))
    return F.when(c.rlike("^[0-9]+$"), F.regexp_replace(c, "^0+(?=.)", "")).otherwise(c)

def _norm_valor(v):
    v = v.strip()
    return (v.lstrip("0") or "0") if v.isdigit() else v

def expr_conjunto(tipo_regla, cod_indicador="*", paquete=COD_PAQUETE):
    """DDD 5.7.3: (c11 AND c12 ...) OR (c21 AND ...). Nulo no cumple ninguna condición."""
    filas = (prm_condicion.filter((F.col("cod_paquete") == paquete) & (F.col("tipo_regla") == tipo_regla) &
                                  (F.col("cod_indicador") == cod_indicador)).orderBy("cod_regla", "secuencia").collect())
    reglas = {}
    for r in filas:
        tipo = variables_ok[r.nombre_variable].tipo_variable
        col = F.col(r.nombre_variable)                     # ya normalizada en el modelo lógico
        vals = [_norm_valor(v) for v in r.valores.split(",")]
        if   r.operador == "IN":     e = col.isin(vals)
        elif r.operador == "NOT_IN": e = ~col.isin(vals)
        elif r.operador == "GT":     e = col > float(vals[0])
        elif r.operador == "LT":     e = col < float(vals[0])
        reglas.setdefault(r.cod_regla, []).append(col.isNotNull() & e)
    if not reglas:
        return None
    return functools.reduce(lambda a, b: a | b, [functools.reduce(lambda a, b: a & b, es) for es in reglas.values()])

def corte_menos(corte, n=1):
    y, m = int(corte[:4]), int(corte[4:])
    t = y * 12 + (m - 1) - n
    return "%04d%02d" % (t // 12, t % 12 + 1)

def evaluar(df):
    """Aplica operador del catálogo, umbral por clasificación y M11 (DDD 5.7.1, Anexo A)."""
    cat = spark.createDataFrame([(k, v.operador, v.nivel, v.sentido) for k, v in CATALOGO.items()],
                                "cod_indicador string, operador string, nivel string, sentido string")
    d = (df.join(cat, "cod_indicador")
           .join(prm_indicador_paquete, ["cod_indicador", "cod_paquete"])
           .withColumn("umbral_aplicado", F.when(F.col("clasificacion_corte") == 1, F.col("umbral_pequena"))
                                           .when(F.col("clasificacion_corte") == 2, F.col("umbral_mediana"))
                                           .otherwise(F.col("umbral_grande"))))
    v, u = F.col("valor_indicador"), F.col("umbral_aplicado")
    d = d.withColumn("supera_umbral",
            F.when(v.isNull(), F.lit(None))
             .when(F.col("operador") == "ABS_GE", F.abs(v) >= u)
             .when(F.col("operador") == "GE", v >= u))
    d = d.withColumn("desviacion_umbral",
            F.when(F.col("sentido") == "PORCENTAJE_BILATERAL", F.abs(v) - u)
             .when(F.col("sentido") == "PORCENTAJE_SUPERIOR", v - u)
             .when(F.col("sentido") == "NUMERICO_SUPERIOR", F.col("registros_afectados") - u))
    return d.drop("umbral_pequena", "umbral_mediana", "umbral_grande", "operador", "nivel")

# ===================== 7. CÁLCULO DE INDICADORES -> FACT_REGLA =====================
act = base.filter(F.col("corte_registro") == CORTE_ACTUAL)            # registros del corte actual
abiertas_act = act.filter("es_abierto")
abiertas_todas = base.filter("es_abierto")                             # para VOL005 (no actualizadas)
g = ["cod_entidad"]

def ind(df, codigo, evaluados, afectados, medida=None, pct=True):
    """Cada indicador = una agregación por entidad. pct=True -> valor = afectados/evaluados*100."""
    medida = (medida if medida is not None else afectados).cast("double")
    out = df.groupBy(*g).agg(evaluados.alias("registros_evaluados"), afectados.alias("registros_afectados"),
                             medida.alias("medida_actual"))
    valor = (F.col("registros_afectados") / F.col("registros_evaluados") * 100) if pct else F.col("registros_afectados")
    return out.withColumn("cod_indicador", F.lit(codigo)).withColumn("valor_indicador", valor.cast("double"))

def si(c):  # cuenta registros que cumplen c
    return F.sum(F.when(c, 1).otherwise(0))

n = F.count(F.lit(1))
resultados = [
    # --- Indicadores SIN corte anterior: resultado real hoy ---
    ind(abiertas_todas, "VOL005", n, si(F.col("corte_registro") < CORTE_ACTUAL)),
    ind(abiertas_act,   "MON007", n, si(F.col("VALOR_SALDO") == 0)),
    ind(abiertas_act,   "MON009", n, si(F.col("VALOR_CUOTA") == 0)),
    ind(abiertas_act,   "MON010", n, si(F.col("VALOR_REAL_PAGADO") == 0)),
    ind(act,            "EST001", n, si(functools.reduce(lambda a, b: a | b,
                                    [F.col(k).isNull() for k, r in variables_ok.items() if r.valida_nulo == 1])), pct=False),
    # --- Indicadores CON corte anterior: medida actual; el % sale de comparar con el corte anterior ---
    ind(abiertas_act,   "VOL001", n, n),
    ind(abiertas_act,   "VOL002", n, si((F.col("EDAD_MORA") > 0) & (F.col("VALOR_MORA") > 0))),   # criterio fijo P-38
    ind(abiertas_act,   "MON001", n, n, medida=F.sum("VALOR_SALDO").cast("double")),
]
fact = functools.reduce(lambda a, b: a.unionByName(b), resultados).withColumn("cod_paquete", F.lit(COD_PAQUETE))

# Corte anterior: lo guarda el propio motor (la fuente se reescribe cada noche y no tiene historia)
def existe(ruta):
    p = spark._jvm.org.apache.hadoop.fs.Path(ruta)
    return p.getFileSystem(spark._jsc.hadoopConfiguration()).exists(p)

RUTA_FACT = RUTA_SALIDA + "/fact_regla"
if existe(RUTA_FACT):
    ant = (spark.read.format(FORMATO_SALIDA).load(RUTA_FACT)
           .filter(F.col("id_corte") == CORTE_ANTERIOR)
           .select("cod_entidad", "cod_paquete", "cod_indicador", F.col("medida_actual").alias("medida_anterior")))
else:
    ant = spark.createDataFrame([], "cod_entidad string, cod_paquete string, cod_indicador string, medida_anterior double")
fact = fact.join(ant, ["cod_entidad", "cod_paquete", "cod_indicador"], "left")

usa_ant = [k for k, v in CATALOGO.items() if v.corte_anterior]
es_var = F.col("cod_indicador").isin(usa_ant)
fact = (fact.withColumn("medida_anterior", F.when(es_var, F.col("medida_anterior")))
            .withColumn("valor_indicador", F.when(es_var,
                 F.when(F.col("medida_anterior") > 0,
                        (F.col("medida_actual") - F.col("medida_anterior")) / F.col("medida_anterior") * 100))
                 .otherwise(F.col("valor_indicador")))
            .withColumn("estado_evaluacion", F.when(es_var & F.col("medida_anterior").isNull(), "LINEA_BASE_SIN_CORTE_ANTERIOR")
                                              .when(es_var & (F.col("medida_anterior") == 0), "ERROR_CARGUE_ANTERIOR")
                                              .otherwise("EVALUADO")))

# Clasificación de la entidad por volumen del corte (Q-01)
vol = act.groupBy(*g).agg(F.count(F.lit(1)).alias("volumen_registros"))
cla = prm_clasificacion.select("cod_clasificacion", "volumen_min", "volumen_max", "meta_ratio")
vol = (vol.join(F.broadcast(cla), (F.col("volumen_registros") >= F.col("volumen_min")) &
               (F.col("volumen_max").isNull() | (F.col("volumen_registros") <= F.col("volumen_max"))))
          .select(*g, "volumen_registros", F.col("cod_clasificacion").alias("clasificacion_corte"), "meta_ratio"))

fact_regla = (evaluar(fact.join(vol, g, "left"))
              .withColumn("id_corte", F.lit(CORTE_ACTUAL))
              .withColumn("id_ejecucion", F.lit(ID_EJECUCION))
              .select("id_corte", "cod_entidad", "cod_paquete", "cod_indicador", "clasificacion_corte", "meta_ratio",
                      "volumen_registros", "registros_evaluados", "registros_afectados", "medida_actual",
                      "medida_anterior", F.round("valor_indicador", 4).alias("valor_indicador"), "umbral_aplicado",
                      "supera_umbral", F.round("desviacion_umbral", 4).alias("desviacion_umbral"), "sentido",
                      "estado_evaluacion", "id_ejecucion"))

def escribir(df, ruta):
    w = df.write.format(FORMATO_SALIDA).partitionBy("id_corte")
    if existe(ruta) and FORMATO_SALIDA == "delta":
        w.mode("overwrite").option("replaceWhere", "id_corte = '%s'" % CORTE_ACTUAL).save(ruta)
    elif existe(ruta):
        spark.conf.set("spark.sql.sources.partitionOverwriteMode", "dynamic"); w.mode("overwrite").save(ruta)
    else:
        w.mode("overwrite").save(ruta)

escribir(fact_regla, RUTA_FACT)
